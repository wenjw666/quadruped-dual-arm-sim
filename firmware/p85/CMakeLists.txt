cmake_minimum_required(VERSION 3.16)
project(p85_rt0_foundation C)
message(STATUS "P85 Linux validation build; this does NOT generate R0 firmware")
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(P85_SANITIZE "Enable host address and undefined behavior checks" OFF)
add_compile_options(-Wall -Wextra -Werror -Wpedantic -Wshadow -Wstrict-prototypes)
if(P85_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(p85 STATIC src/p85_protocol.c src/p85_control.c src/p85_wire.c rt0/p85_runtime.c)
target_include_directories(p85 PUBLIC include)
target_link_libraries(p85 PUBLIC m)
add_executable(p85_tests tests/test_p85.c)
target_link_libraries(p85_tests PRIVATE p85)
add_executable(p85d host/p85d.c)
target_link_libraries(p85d PRIVATE p85)
add_executable(p85_runtime_tests tests/test_runtime.c)
target_link_libraries(p85_runtime_tests PRIVATE p85)
enable_testing()
add_test(NAME p85_core COMMAND p85_tests)
add_test(NAME p85_runtime COMMAND p85_runtime_tests)

find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
 add_test(NAME p85_r2_handoff COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_r2_handoff.py)
 add_test(NAME p85_handoff_flash_guard COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_handoff_flash_guard.py)
 add_test(NAME p85_can_trace COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_can_trace.py)
 add_test(NAME p85_link_report COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_can_link_report.py)
 add_test(NAME p85_loopback_report COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_can_loopback_report.py)
 add_test(NAME p85_aon_state COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_aon_state.py)
 add_test(NAME p85_host COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_host.py ${CMAKE_CURRENT_BINARY_DIR})
 set_tests_properties(p85_host PROPERTIES TIMEOUT 60)
endif()

find_path(BST_MSGBX_INCLUDE ipc_trans_layer.h PATHS /usr/bst/include/msgbx NO_DEFAULT_PATH)
find_library(BST_MSGBX_LIBRARY msgbx PATHS /usr/bst/lib NO_DEFAULT_PATH)
if(BST_MSGBX_INCLUDE AND BST_MSGBX_LIBRARY)
 message(STATUS "P85: official Linux MsgBox adapter available; R0 firmware is built separately with Hanhai SDK")
 add_library(p85_bst_header_check OBJECT rt0/p85_bst_msgbox.c)
 target_include_directories(p85_bst_header_check PRIVATE include rt0)
 target_include_directories(p85_bst_header_check SYSTEM PRIVATE ${BST_MSGBX_INCLUDE})
 add_executable(p85_bst_tests tests/test_bst_msgbox.c $<TARGET_OBJECTS:p85_bst_header_check>)
 target_include_directories(p85_bst_tests PRIVATE rt0)
 target_include_directories(p85_bst_tests SYSTEM PRIVATE ${BST_MSGBX_INCLUDE})
 target_link_libraries(p85_bst_tests PRIVATE p85)
 add_test(NAME p85_bst_adapter COMMAND p85_bst_tests)
 add_executable(p85_hanhai_receive_tests tests/test_hanhai_receive.c
   rt0/hanhai/p85_hanhai_msgbox.c $<TARGET_OBJECTS:p85_bst_header_check>)
 target_include_directories(p85_hanhai_receive_tests PRIVATE rt0 rt0/hanhai)
 target_include_directories(p85_hanhai_receive_tests SYSTEM PRIVATE ${BST_MSGBX_INCLUDE})
 target_link_libraries(p85_hanhai_receive_tests PRIVATE p85)
 add_test(NAME p85_hanhai_receive COMMAND p85_hanhai_receive_tests)
 add_library(p85_msgbox SHARED host/p85_msgbox.c)
 target_include_directories(p85_msgbox SYSTEM PRIVATE ${BST_MSGBX_INCLUDE})
 target_link_libraries(p85_msgbox PRIVATE ${BST_MSGBX_LIBRARY} pthread)
 set_target_properties(p85_msgbox PROPERTIES BUILD_RPATH /usr/bst/lib)
 add_executable(p85_ipc_probe host/p85_ipc_probe.c)
 target_link_libraries(p85_ipc_probe PRIVATE p85_msgbox)
else()
 message(STATUS "P85: vendor MsgBox headers/library absent; portable core and software simulation only")
endif()
