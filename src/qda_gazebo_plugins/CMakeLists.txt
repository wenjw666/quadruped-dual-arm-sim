cmake_minimum_required(VERSION 3.8)
project(qda_gazebo_plugins)
find_package(ament_cmake REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(qda_interfaces REQUIRED)

add_library(qda_grasp_plugin SHARED src/grasp_plugin.cpp)
target_include_directories(qda_grasp_plugin PUBLIC ${GAZEBO_INCLUDE_DIRS})
target_compile_features(qda_grasp_plugin PUBLIC cxx_std_17)
ament_target_dependencies(qda_grasp_plugin gazebo_ros geometry_msgs sensor_msgs nav_msgs rclcpp qda_interfaces)
target_link_libraries(qda_grasp_plugin ${GAZEBO_LIBRARIES})
install(TARGETS qda_grasp_plugin LIBRARY DESTINATION lib)
ament_environment_hooks("${CMAKE_CURRENT_SOURCE_DIR}/env-hooks/qda_plugin_path.dsv")
ament_package()
